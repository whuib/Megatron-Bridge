# Format: <name> <mode> <layout args>. Language TP > 1 always uses sequence parallelism.

# Native: vision runs inside the first language pipeline stage with the language TP and DP.

nat_t4p1 native --language-tp 4 --language-pp 1 --sequence-parallel
